cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD waveshare_rp2040_zero)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(proxy_ctrl C CXX ASM)
pico_sdk_init()
add_executable(proxy_ctrl main.c)
target_compile_options(proxy_ctrl PRIVATE -Wall -Wextra)
target_link_libraries(proxy_ctrl PRIVATE pico_stdlib)
pico_enable_stdio_usb(proxy_ctrl 1)
pico_enable_stdio_uart(proxy_ctrl 0)
pico_add_extra_outputs(proxy_ctrl)
